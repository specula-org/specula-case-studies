+    let sms: Vec<_> = (0..3).map(|_| Arc::new(Accumulator::new())).collect();
+    let replicas: Vec<_> = sms
+        .iter()
+        .map(|sm| {
+            let id = config.add_replica();
+            Replica::new(
+                id,
+                config.clone(),
+                sm.clone(),
+                client_tx.clone(),
+                replica_tx.clone(),
+            )
+        })
+        .collect();
+    let tick_with = |deliver: &dyn Fn(usize, &Message<Op>) -> bool| {
+        while !replica_rx.is_empty() {
+            let (replica_id, message) = replica_rx.recv().unwrap();
+            if deliver(replica_id, &message) {
+                replicas[replica_id].on_message(message);
+            }
+        }
+    };
+    let idle_all = || {
+        for replica in &replicas {
+            replica.on_idle();
+        }
+    };
+    // Replica 0 has crashed: it gets nothing and runs no idle logic.
+    let idle_without_0 = || {
+        replicas[1].on_idle();
+        replicas[2].on_idle();
+    };
+    let client = Client::new(0, config, replica_tx);
+
+    // Two ops commit everywhere in view 0.
+    client.on_request(Op::Add(10), Box::new(|_| {}));
+    client.on_request(Op::Add(20), Box::new(|_| {}));
+    tick_with(&|_, _| true);
+    idle_all();
+    tick_with(&|_, _| true);
+    for sm in &sms {
+        assert_eq!(30, *sm.accumulator.lock());
+    }
+    assert_eq!(2, client_rx.try_iter().count());
+
+    // A third op reaches the backups, but the primary crashes before it
+    // sees their acknowledgements.
+    let request_number = client.on_request(Op::Add(30), Box::new(|_| {}));
+    tick_with(&|replica_id, message| replica_id != 0 || matches!(message, Message::Request { .. }));
+    assert_eq!(3, replicas[1].op_number());
+    assert_eq!(3, replicas[2].op_number());
+    assert_eq!(2, replicas[0].commit_number());
+
+    // The backups stop hearing from the primary and change view.
+    for _ in 0..10 {
+        idle_without_0();
+        tick_with(&|replica_id, _| replica_id != 0);
+    }
+    assert_eq!(1, replicas[1].view_number());
+    assert_eq!(1, replicas[2].view_number());
+    assert!(replicas[1].is_primary());
+    assert!(!replicas[2].is_primary());
+    // The new primary commits the op it found and replies to the client.
+    assert_eq!(3, replicas[1].commit_number());
+    assert_eq!(3, replicas[2].commit_number());
+    assert_eq!(60, *sms[1].accumulator.lock());
+    assert_eq!(60, *sms[2].accumulator.lock());
+    let replies: Vec<_> = client_rx.try_iter().collect();
+    assert_eq!(1, replies.len());
+    assert_eq!(request_number, replies[0].request_number);
+    assert_eq!(1, replies[0].view_number);
+
+    // The client learns the view from the reply and sends the next request
+    // to the new primary.
+    client.on_reply(replies[0].request_number, replies[0].view_number);
+    client.on_request(Op::Add(40), Box::new(|_| {}));
+    tick_with(&|replica_id, _| replica_id != 0);
+    idle_without_0();
+    tick_with(&|replica_id, _| replica_id != 0);
+    assert_eq!(4, replicas[1].commit_number());
+    assert_eq!(4, replicas[2].commit_number());
+    assert_eq!(100, *sms[1].accumulator.lock());
+    assert_eq!(100, *sms[2].accumulator.lock());
+    assert_eq!(1, client_rx.try_iter().count());
+    // The crashed primary never moved.
+    assert_eq!(0, replicas[0].view_number());
+    assert_eq!(30, *sms[0].accumulator.lock());
+}
